CREATE TABLE IF NOT EXISTS contact_log (    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY NOT NULL,    customer_id BIGINT NOT NULL,    channel VARCHAR(30) NOT NULL,    note LONGVARCHAR,    contacted_at TIMESTAMP DEFAULT now() NOT NULL);
CREATE TABLE IF NOT EXISTS customer_category (    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY NOT NULL,    code VARCHAR(30) NOT NULL,    name VARCHAR(100) NOT NULL,    rank INT DEFAULT 0 NOT NULL,    is_active BOOLEAN DEFAULT TRUE NOT NULL);
CREATE TABLE IF NOT EXISTS customer (    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY NOT NULL,    category_id BIGINT,    name VARCHAR(100) NOT NULL,    email VARCHAR(255) NOT NULL,    phone VARCHAR(30),    created_at TIMESTAMP DEFAULT now() NOT NULL,    updated_at TIMESTAMP DEFAULT now() NOT NULL);

import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

import com.example.customer.service.AddressService;

@RestController
@RequestMapping("/api/address")
public class AddressController {

    @Autowired
    private AddressService service;

    @GetMapping
    public List<Map<String, Object>> select(@RequestParam Map<String, Object> params) {
        return service.select_address_datalist_map(params);
    }

    @PostMapping
    public int save(@RequestBody List<Map<String, Object>> rows) {
        return service.save_address_datalist_map(rows);
    }
}
